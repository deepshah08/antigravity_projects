Given a string s consisting of lowercase letters and parentheses '(' and ')'.A string is considered valid if:Every opening parenthesis '(' has a corresponding closing parenthesis ')'.Parentheses are properly nested.Remove the minimum number of invalid parentheses from s so that the resulting string becomes valid.  Return all the possible distinct valid strings in lexicographically sorted order.Examples:Input: s = "()())()"Output: ["(())()", "()()()"]Explanation: The string "()())()" has one extra ')', making it invalid. By removing one ')', we can make it valid in two ways:Remove the 1st index ')' -> "(())()"Remove the 4th index ')' -> "()()()"Both are valid and require the minimum removals.Input: s = "(a)())()"Output: ["(a())()", "(a)()()"]Explanation: We remove one ')' (minimum removals) to make it valid. Possible valid results:Remove a ')' -> "(a())()"Remove another ')' -> "(a)()()"Input: s = ")("Output: [""]Explanation: The string ")(" is invalid. Removing both parentheses (minimum removals) gives an empty string "", which is valid.Try It YourselfTable of ContentBreadth First Search (BFS) - O(2 ^ n) Time and O(2 ^ n) SpaceDepth First Search (DFS) - O(2 ^ n) Time and O(2 ^ n) SpaceBreadth First Search (BFS) - O(2 ^ n) Time and O(2 ^ n) SpaceThe idea is to remove parentheses one by one and check whether the string is valid.     #include <bits/stdc++.h>
using namespace std;

// Check whether the given string has valid parentheses.
bool isValid(string s)
{
    int count = 0;

    // Traverse the string.
    for (char c : s)
    {

        // Increment for an opening parenthesis.
        if (c == '(')
            count++;

        // Decrement for a closing parenthesis.
        else if (c == ')')
        {
            count--;

            // More closing parentheses than opening ones.
            if (count < 0)
                return false;
        }
    }

    // Valid only when all opening parentheses are matched.
    return count == 0;
}

// Remove the minimum number of invalid parentheses.
vector<string> validParenthesis(string &s)
{

    // Store strings that have already been visited.
    unordered_set<string> visited;

    // Queue for BFS.
    queue<string> q;

    // Store all valid strings.
    vector<string> res;

    // Push the original string.
    q.push(s);

    // Mark the original string as visited.
    visited.insert(s);

    // Indicates that a valid string has been found.
    bool found = false;

    // Perform BFS.
    while (!q.empty())
    {

        // Get the current string.
        string curr = q.front();
        q.pop();

        // Check whether the current string is valid.
        if (isValid(curr))
        {
            res.push_back(curr);
            found = true;
        }

        // Do not generate the next level after
        // finding a valid string.
        if (found)
            continue;

        // Try removing each parenthesis.
        for (int i = 0; i < curr.length(); i++)
        {

            // Skip normal characters.
            if (curr[i] != '(' && curr[i] != ')')
                continue;

            // Remove the parenthesis at index i.
            string next = curr.substr(0, i) + curr.substr(i + 1);

            // Process only unvisited strings.
            if (visited.find(next) == visited.end())
            {
                visited.insert(next);
                q.push(next);
            }
        }
    }

    // Sort the valid strings lexicographically.
    sort(res.begin(), res.end());

    return res;
}

        // Queue for BFS.

        q.Enqueue(s);

            // finding a valid string.

            for (int i = 0; i < curr.Length; i++) {

                if (curr[i] != '(' && curr[i] != ')')

                string next = curr.Substring(0, i)
                              + curr.Substring(i + 1);

                if (!visited.Contains(next)) {
                    visited.Add(next);
                    q.Enqueue(next);
                }
            }
        }

        // Sort the valid strings lexicographically.
        res.Sort();

        return res;
    }

    #include <bits/stdc++.h>
using namespace std;

// Recursive function to find all valid strings.
void findValid(string str, int index, int open, int close, int pair, string cur, vector<string> &res,
               unordered_set<string> &st)
{
    // If the end of the string is reached.
    if (index == str.size())
    {
        // Check if all invalid parentheses are removed
        // and no unmatched opening parenthesis is left.
        if (open == 0 && close == 0 && pair == 0)
        {
            // Add only distinct strings.
            if (st.find(cur) == st.end())
            {
                res.push_back(cur);
                st.insert(cur);
            }
        }

        return;
    }

    // If the current character is not a parenthesis.
    if (str[index] != '(' && str[index] != ')')
    {
        // Keep the character.
        findValid(str, index + 1, open, close, pair, cur + str[index], res, st);
    }
    else
    {
        // If the current character is an opening parenthesis.
        if (str[index] == '(')
        {
            // Remove the current parenthesis.
            if (open > 0)
            {
                findValid(str, index + 1, open - 1, close, pair, cur, res, st);
            }

            // Keep the current parenthesis.
            findValid(str, index + 1, open, close, pair + 1, cur + str[index], res, st);
        }
        else
        {
            // Remove the current parenthesis.
            if (close > 0)
            {
                findValid(str, index + 1, open, close - 1, pair, cur, res, st);
            }

            // Keep the current parenthesis if it has
            // a matching opening parenthesis.
            if (pair > 0)
            {
                findValid(str, index + 1, open, close, pair - 1, cur + str[index], res, st);
            }
        }
    }
}

    function findValid(str, index, open, close, pair, cur, res,
                   st)
{
    // If the end of the string is reached.
    if (index === str.length) {
        // Check if all invalid parentheses are removed
        // and no unmatched opening parenthesis is left.
        if (open === 0 && close === 0 && pair === 0) {
            // Add only distinct strings.
            if (!st.has(cur)) {
                res.push(cur);
                st.add(cur);
            }
        }

        return;
    }

    // If the current character is not a parenthesis.
    if (str[index] !== "(" && str[index] !== ")") {
        // Keep the character.
        findValid(str, index + 1, open, close, pair,
                  cur + str[index], res, st);
    }
    else {
        // If the current character is an opening
        // parenthesis.
        if (str[index] === "(") {
            // Remove the current parenthesis.
            if (open > 0) {
                findValid(str, index + 1, open - 1, close,
                          pair, cur, res, st);
            }

            // Keep the current parenthesis.
            findValid(str, index + 1, open, close, pair + 1,
                      cur + str[index], res, st);
        }
        else {
            // Remove the current parenthesis.
            if (close > 0) {
                findValid(str, index + 1, open, close - 1,
                          pair, cur, res, st);
            }

            // Keep the current parenthesis if it has
            // a matching opening parenthesis.
            if (pair > 0) {
                findValid(str, index + 1, open, close,
                          pair - 1, cur + str[index], res,
                          st);
            }
        }
    }
}

function validParenthesis(str)
{
    let result = [];
    let st = new Set();

    // Count invalid opening and closing parentheses.
    let open = 0, close = 0;

    // Find the minimum parentheses to remove.
    for (let c of str) {
        // Count opening parentheses.
        if (c === "(") {
            open++;
            // Match a closing parenthesis with an opening
            // one.
        }
        else if (c === ")") {
            if (open !== 0) {
                open--;
            }
            else {
                close++;
            }
        }
    }

    // Generate all valid strings.
    findValid(str, 0, open, close, 0, "", result, st);

    // Sort the result lexicographically.
    result.sort();

    return result;
}
