# Check if Two Trees are Identical - GeeksforGeeks

Given two binary trees with their root nodes r1 and r2, return true if both of them are identical, otherwise return false.Note: Two trees are identical when they have the same data and the arrangement of the data is also same.Examples:Input: Output: trueExplanation: Trees are identical, having same node structure and node values.Input:        Output: falseExplanation: Trees are not identical, both have different node structure.Try It YourselfTable of ContentRecursion (DFS) - O(n) Time and O(h) SpaceLevel Order Traversal (BFS) - O(n) Time and O(w) SpaceMorris Traversal - O(n) Time and O(1) SpaceRecursion (DFS) - O(n) Time and O(h) SpaceThe idea is to compare both trees simultaneously using DFS. At every corresponding node, check its value and recursively compare its left and right subtrees.Working of Approach:If both nodes are NULL, the corresponding parts are identical.If only one node is NULL, the tree structures are different.If the data of both nodes is different, return false.Recursively compare the left and right subtrees.Return true only if both subtrees are identical.
            C++

    #include <iostream>
using namespace std;

class Node
{
  public:
    int data;
    Node *left;
    Node *right;

    Node(int x)
    {
        data = x;
        left = right = nullptr;
    }
};

bool isIdentical(Node *r1, Node *r2)
{
    // If both nodes are NULL, they are identical
    if (r1 == nullptr && r2 == nullptr)
        return true;

    // If only one node is NULL, they are not identical
    if (r1 == nullptr || r2 == nullptr)
        return false;

    // Check data and compare left and right subtrees
    return (r1->data == r2->data) && isIdentical(r1->left, r2->left) && isIdentical(r1->right, r2->right);
}

int main()
{

    // Create first tree
    Node *r1 = new Node(1);
    r1->left = new Node(2);
    r1->right = new Node(3);
    r1->left->left = new Node(4);

    // Create second tree
    Node *r2 = new Node(1);
    r2->left = new Node(2);
    r2->right = new Node(3);
    r2->left->left = new Node(4);

    cout << boolalpha << isIdentical(r1, r2) << endl;

    return 0;
}

    #include <iostream>
#include <queue>
using namespace std;

class Node
{
  public:
    int data;
    Node *left;
    Node *right;

    Node(int x)
    {
        data = x;
        left = right = nullptr;
    }
};

bool isIdentical(Node *r1, Node *r2)
{
    queue<Node *> q1, q2;

    // Add the roots of both trees
    q1.push(r1);
    q2.push(r2);

    while (!q1.empty() && !q2.empty())
    {
        Node *a = q1.front();
        Node *b = q2.front();

        q1.pop();
        q2.pop();

        // If both nodes are NULL, continue
        if (a == nullptr && b == nullptr)
            continue;

        // If only one node is NULL, they are not identical
        if (a == nullptr || b == nullptr)
            return false;

        // If data is different, trees are not identical
        if (a->data != b->data)
            return false;

        // Compare corresponding children
        q1.push(a->left);
        q1.push(a->right);

        q2.push(b->left);
        q2.push(b->right);
    }

    // Both queues should be empty together
    return q1.empty() && q2.empty();
}

typedef struct Queue
{
    Node **array;
    int front, rear, capacity;
} Queue;

Queue *createQueue(int capacity)
{
    Queue *queue = (Queue *)malloc(sizeof(Queue));
    queue->array = (Node **)malloc(sizeof(Node *) * capacity);
    queue->capacity = capacity;
    queue->front = queue->rear = 0;
    return queue;
}

bool isQueueEmpty(Queue *queue)
{
    return queue->rear == queue->front;
}

void enqueue(Queue *queue, Node *node)
{
    if (queue->rear == queue->capacity)
        return;
    queue->array[queue->rear++] = node;
}

Node *dequeue(Queue *queue)
{
    if (isQueueEmpty(queue))
        return NULL;
    Node *node = queue->array[queue->front++];
    return node;
}

bool isIdentical(Node *r1, Node *r2)
{
    Queue *q1 = createQueue(100);
    Queue *q2 = createQueue(100);

    // Add the roots of both trees
    enqueue(q1, r1);
    enqueue(q2, r2);

    while (!isQueueEmpty(q1) && !isQueueEmpty(q2))
    {
        Node *a = dequeue(q1);
        Node *b = dequeue(q2);

        // If both nodes are NULL, continue
        if (a == NULL && b == NULL)
            continue;

        // If only one node is NULL, they are not identical
        if (a == NULL || b == NULL)
            return false;

        // If data is different, trees are not identical
        if (a->data != b->data)
            return false;

        // Compare corresponding children
        enqueue(q1, a->left);
        enqueue(q1, a->right);

        enqueue(q2, b->left);
        enqueue(q2, b->right);
    }

    // Both queues should be empty together
    return isQueueEmpty(q1) && isQueueEmpty(q2);
}

Along with each node's value, we compare whether its left and right children exist, which preserves the complete tree structure.Working of Approach:Traverse both trees using Morris preorder traversal.For every visited node, compare its data and the existence of its left and right children.Morris traversal temporarily creates a thread from the inorder predecessor to the current node.Remove every temporary thread after the left subtree is processed, restoring the original tree.If all corresponding nodes match and both traversals finish together, the trees are identical.

    #include <iostream>
using namespace std;

class Node
{
  public:
    int data;
    Node *left;
    Node *right;

    Node(int x)
    {
        data = x;
        left = right = nullptr;
    }
};

// Returns the next node in Morris preorder traversal.
// Also stores the node's data and child structure.
bool getNext(Node *&curr, int &data, bool &hasLeft, bool &hasRight)
{

    while (curr != nullptr)
    {

        // If there is no left subtree
        if (curr->left == nullptr)
        {
            data = curr->data;
            hasLeft = false;
            hasRight = (curr->right != nullptr);

            curr = curr->right;
            return true;
        }

        // Find the inorder predecessor
        Node *pred = curr->left;

        while (pred->right != nullptr && pred->right != curr)
            pred = pred->right;

        // Create a temporary thread
        if (pred->right == nullptr)
        {
            data = curr->data;
            hasLeft = true;
            hasRight = (curr->right != nullptr);

            pred->right = curr;
            curr = curr->left;

            return true;
        }

        // Remove the temporary thread
        pred->right = nullptr;
        curr = curr->right;
    }

    return false;
}

bool isIdentical(Node *r1, Node *r2)
{

    Node *curr1 = r1;
    Node *curr2 = r2;

    while (true)
    {

        int data1, data2;
        bool left1, right1;
        bool left2, right2;

        bool found1 = getNext(curr1, data1, left1, right1);
        bool found2 = getNext(curr2, data2, left2, right2);

        // If both traversals are complete
        if (!found1 && !found2)
            return true;

        // If only one traversal is complete
        if (found1 != found2)
            return false;

        // Compare node data and tree structure
        if (data1 != data2 || left1 != left2 || right1 != right2)
            return false;
    }
}

        boolean found;
        int data;
        boolean hasLeft;
        boolean hasRight;

        Result(boolean found, int data, boolean hasLeft,
               boolean hasRight)
        {
            this.found = found;
            this.data = data;
            this.hasLeft = hasLeft;
            this.hasRight = hasRight;
        }
    }

    // Returns the next node in Morris preorder traversal
    // and stores its data and child structure
    static Result getNext(Node[] curr)
    {

        while (curr[0] != null) {

            // If there is no left subtree
            if (curr[0].left == null) {
                int data = curr[0].data;
                boolean hasLeft = false;
                boolean hasRight = curr[0].right != null;

                curr[0] = curr[0].right;

                return new Result(true, data, hasLeft,
                                  hasRight);
            }

            // Find the inorder predecessor
            Node pred = curr[0].left;

            while (pred.right != null
                   && pred.right != curr[0])
                pred = pred.right;

            // Create a temporary thread
            if (pred.right == null) {
                int data = curr[0].data;
                boolean hasLeft = true;
                boolean hasRight = curr[0].right != null;

                pred.right = curr[0];
                curr[0] = curr[0].left;

                return new Result(true, data, hasLeft,
                                  hasRight);
            }

            // Remove the temporary thread
            pred.right = null;
            curr[0] = curr[0].right;
        }

        return new Result(false, 0, false, false);
    }

    static boolean isIdentical(Node r1, Node r2)
    {

        Node[] curr1 = { r1 };
        Node[] curr2 = { r2 };

        while (true) {

            Result res1 = getNext(curr1);
            Result res2 = getNext(curr2);

            // If both traversals are complete
            if (!res1.found && !res2.found)
                return true;

            // If only one traversal is complete
            if (res1.found != res2.found)
                return false;

            // Compare node data and tree structure
            if (res1.data != res2.data
                || res1.hasLeft != res2.hasLeft
                || res1.hasRight != res2.hasRight)
                return false;
        }
    }

    public static void main(String[] args)
    {

        // Create first tree
        Node r1 = new Node(1);
        r1.left = new Node(2);
        r1.right = new Node(3);
        r1.left.left = new Node(4);

        // Create second tree
        Node r2 = new Node(1);
        r2.left = new Node(2);
        r2.right = new Node(3);
        r2.left.left = new Node(4);

        // Check whether both trees are identical
        System.out.println(isIdentical(r1, r2));
    }
}

        if curr.left is None:
            data = curr.data
            hasLeft = False
            hasRight = curr.right is not None

            curr = curr.right
            return curr, True, data, hasLeft, hasRight

        # Find the inorder predecessor
        pred = curr.left

        while pred.right is not None and pred.right != curr:
            pred = pred.right

        # Create a temporary thread
        if pred.right is None:
            data = curr.data
            hasLeft = True
            hasRight = curr.right is not None

            pred.right = curr
            curr = curr.left

            return curr, True, data, hasLeft, hasRight

        # Remove the temporary thread
        pred.right = None
        curr = curr.right

    return curr, False, None, False, False

    static bool GetNext(ref Node curr, out int data,
                        out bool hasLeft, out bool hasRight)
    {

        while (curr != null) {

            // If there is no left subtree
            if (curr.left == null) {
                data = curr.data;
                hasLeft = false;
                hasRight = curr.right != null;

                curr = curr.right;
