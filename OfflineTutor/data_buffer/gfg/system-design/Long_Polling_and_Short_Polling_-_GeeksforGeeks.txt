# Long Polling and Short Polling - GeeksforGeeks

Polling is a communication technique in which a client repeatedly contacts a server to check whether new data is available. It is commonly used when applications need periodic or near real-time updates without maintaining a persistent bidirectional connection such as WebSocket.Short Polling: Sends requests periodically and receives an immediate response.Long Polling: Sends a request that the server may keep open until new data is available or a timeout occurs.Example: A news app refreshing every 10 seconds to fetch the latest headlines from the server.Short PollingShort polling is a technique where the client continuously sends requests to the server at fixed intervals. The server immediately responds—either with data if available or an empty response if not. This cycle keeps repeating, allowing the client to check for updates regularly.Simple to implement using standard HTTP/AJAX requests without complex setup.Can cause unnecessary requests and increase server load due to frequent polling even when no new data is available.Example:  A chat application that sends a request to the server every 5 seconds to check for new messages, even if no new messages have arrived.     <h1>GeeksforGeeks</h1>
    <h3>Long Polling using AJAX</h3>

    <button onclick="startPolling()">Start Polling</button>
    <button onclick="stopPolling()">Stop Polling</button>

    <p id="output">Waiting for updates...</p>

    <script>
        let isPolling = false;
        let activeRequest = null;

        function startPolling() {
            if (isPolling) return;

            isPolling = true;
            loadInformation();
        }

        function stopPolling() {
            isPolling = false;

            if (activeRequest) {
                activeRequest.abort();
                activeRequest = null;
            }
        }

        function loadInformation() {
            if (!isPolling) return;

            const request = new XMLHttpRequest();
            activeRequest = request;

            request.open("GET", "/updates", true);
            request.timeout = 35000;

            request.onload = function () {
                activeRequest = null;

                if (this.status === 200) {
                    const data = JSON.parse(this.responseText);

                    document.getElementById("output").textContent =
                        data.message;
                }

                // Immediately start the next long-poll request
                if (isPolling) {
                    loadInformation();
                }
            };

            request.onerror = function () {
                retryPolling();
            };

            request.ontimeout = function () {
                retryPolling();
            };

            request.send();
        }

        function retryPolling() {
            activeRequest = null;

            if (isPolling) {
                setTimeout(loadInformation, 1000);
            }
        }
    </script>

</body>
</html>

            Node

    const express = require("express");
const app = express();

app.use(express.static("public"));

let clients = [];

app.get("/updates", (req, res) => {
    const client = { res, timer: null };
    clients.push(client);

    // End the long-poll request after 30 seconds
    // if no new data becomes available.
    client.timer = setTimeout(() => {
        clients = clients.filter(c => c !== client);

        if (!res.headersSent) {
            res.json({ message: "No new updates" });
        }
    }, 30000);

    // Remove disconnected clients.
    res.on("close", () => {
        clearTimeout(client.timer);
        clients = clients.filter(c => c !== client);
    });
});

// Simulate new data every 10 seconds.
setInterval(() => {
    const message = {
        message: "New update at " +
                 new Date().toLocaleTimeString()
    };

    const pending = clients;
    clients = [];

    pending.forEach(client => {
        clearTimeout(client.timer);

        if (!client.res.headersSent) {
            client.res.json(message);
        }
    });
}, 10000);

app.listen(3000, () => {
    console.log("Server running on http://localhost:3000");
});
